# ยกเลิกใบขาย (Void Sale) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** ให้ OWNER/BRANCH_MANAGER ยกเลิกใบขาย `CASH`/`EXTERNAL_FINANCE` ที่คีย์ผิดได้จากหน้าจอ โดยคืนเครื่องเข้าสต็อก กลับรายการบัญชี เรียกคืนค่าคอม และเหลือหลักฐานครบ

**Tech Stack:** NestJS + Prisma + PostgreSQL · jest (unit) · vitest `--no-file-parallelism` (integration, DB จริง) · React + shadcn/ui

เพิ่มฟิลด์ใน `ProductHoldSubject`:

```ts
  /**
   * สถานะที่ "รายการต้นทางเป็นคนตั้งไว้เอง" — ใช้เฉพาะ `RESTORE_TO_STOCK`
   *
   * ใบขายสดตั้ง `SOLD_CASH`, ใบขายผ่านไฟแนนซ์ภายนอกตั้ง `SOLD_INSTALLMENT` ⇒ ถ้าสถานะ
   * ปัจจุบันยังตรงกับค่านี้แปลว่าไม่มีใครมาผูกต่อ (ข้ามชั้นตาราง `HELD_STATUS_REMEDY` ซึ่ง
   * จะบล็อก `SOLD_INSTALLMENT` ทุกกรณี) แล้วไปตรวจชั้นสัญญา/จอง/ออเดอร์ต่อตามปกติ
   * ถ้าไม่ตรง = มีรายการอื่นผูกเครื่องนี้ไปแล้ว ⇒ บล็อกทันที
   */
  expectedStatus?: ProductStatus;
```

ในตัวฟังก์ชัน หลังชั้น 0 (`product.deletedAt`) และ **ก่อน** การอ่าน `HELD_STATUS_REMEDY`:

```ts
  if (action === 'RESTORE_TO_STOCK') {
    if (!product.expectedStatus) {
      // ผู้เรียกลืมส่ง — เป็นบั๊กของโค้ด ไม่ใช่ข้อผิดพลาดของผู้ใช้
      throw new Error('RESTORE_TO_STOCK requires expectedStatus');
    }
    if (product.status !== product.expectedStatus) {
      throw new BadRequestException(
        `สินค้าอยู่สถานะ ${productStatusLabel(product.status)} ` +
          `(รายการนี้ตั้งไว้เป็น ${productStatusLabel(product.expectedStatus)}) — ${verb} ` +
          `(${risk}): มีรายการอื่นผูกเครื่องนี้ไปแล้ว จัดการรายการนั้นให้จบก่อน`,
      );
    }
    // สถานะตรง = ข้ามชั้นตารางกลาง ไปตรวจชั้นสัญญา/จอง/ออเดอร์ต่อ
  } else {
    const remedy =
      (action === 'RESTORE_TO_CONTRACT' ? RESTORE_ONLY_HELD_REMEDY[product.status] : undefined) ??
      HELD_STATUS_REMEDY[product.status];
    if (remedy) {
      throw new BadRequestException(
        `สินค้าอยู่สถานะ ${productStatusLabel(product.status)} — ${verb}เพราะยังผูกกับรายการที่เดินอยู่ ` +
          `(${risk}): ${remedy}`,
      );
    }
  }
```

> **หมายเหตุ:** บล็อกเดิม (`const remedy = ...` ถึง `}`) ถูกย้ายเข้า `else` **โดยไม่แก้เนื้อใน** — ต้องเหมือนเดิมทุกไบต์เพื่อให้ 3 action เดิมไม่เปลี่ยนพฤติกรรม

```ts
import { BadRequestException, ConflictException, Injectable, Logger, NotFoundException } from '@nestjs/common';
import { Prisma, ProductStatus } from '@prisma/client';
import * as Sentry from '@sentry/node';
import { PrismaService } from '../../prisma/prisma.service';
import { assertProductNotHeld } from '../../products/product-hold.util';
import { ExchangeCancelReversalTemplate } from '../../journal/cpa-templates/exchange-cancel-reversal.template';
import { validatePeriodOpen } from '../../journal/period-guard.util'; // ← ตรวจชื่อจริงก่อนใช้ (grep 'validatePeriodOpen')

/** สถานะที่ใบขายแต่ละแบบตั้งให้สินค้า — ใช้เป็น expectedStatus ของ guard */
const SALE_TYPE_PRODUCT_STATUS: Readonly<Record<'CASH' | 'EXTERNAL_FINANCE', ProductStatus>> = {
  CASH: ProductStatus.SOLD_CASH,
  EXTERNAL_FINANCE: ProductStatus.SOLD_INSTALLMENT,
};

@Injectable()
export class SaleVoidService {
  private readonly logger = new Logger(SaleVoidService.name);

  constructor(
    private prisma: PrismaService,
    private reversalTemplate: ExchangeCancelReversalTemplate,
  ) {}

  async voidSale(saleId: string, userId: string, reason: string) {
    try {
      return await this.prisma.$transaction(
        async (tx) => this.run(tx, saleId, userId, reason),
        { isolationLevel: 'Serializable' },
      );
    } catch (e) {
      if ((e as { code?: string }).code === 'P2034') {
        this.logger.warn(`[sale-void] write conflict (P2034) on sale ${saleId} — rejecting with 409`);
        Sentry.captureMessage('[sale-void] P2034 write-conflict translated to 409', {
          level: 'warning',
        });
        throw new ConflictException('มีการแก้ไขรายการนี้พร้อมกัน กรุณาลองใหม่อีกครั้ง');
      }
      throw e;
    }
  }

  private async run(tx: Prisma.TransactionClient, saleId: string, userId: string, reason: string) {
    const sale = await tx.sale.findUnique({
      where: { id: saleId },
      select: {
        id: true, saleNumber: true, saleType: true, productId: true, bundleProductIds: true,
        contractId: true, onlineOrderId: true, deletedAt: true, voidedById: true, netAmount: true,
      },
    });
    if (!sale) throw new NotFoundException('ไม่พบใบขาย');

    // G1 — idempotency
    if (sale.deletedAt) {
      throw new BadRequestException(
        `ใบขาย ${sale.saleNumber} ถูกยกเลิกไปแล้วเมื่อ ${sale.deletedAt.toLocaleString('th-TH')}`,
      );
    }

    if (sale.saleType === 'INSTALLMENT') {
      throw new BadRequestException(
        'ใบขายผ่อนของเราต้องยกเลิกผ่านเส้นทางยกเลิกสัญญา (เมนู "เอกสารยกเลิกสัญญา") ไม่ใช่ที่นี่',
      );
    }
    // ข้อมูลเพี้ยน — CASH/EXTERNAL_FINANCE ไม่ควรมีสัญญาผูก
    if (sale.contractId) {
      throw new BadRequestException(
        `ใบขาย ${sale.saleNumber} มีสัญญาผูกอยู่ทั้งที่เป็นการขาย${sale.saleType === 'CASH' ? 'สด' : 'ผ่านไฟแนนซ์ภายนอก'} — ข้อมูลผิดปกติ ให้แจ้งผู้ดูแลระบบตรวจสอบก่อน`,
      );
    }
    // G6 — ออเดอร์ออนไลน์
    if (sale.onlineOrderId) {
      throw new BadRequestException(
        `ใบขาย ${sale.saleNumber} มาจากออเดอร์ออนไลน์ — ให้ยกเลิก/คืนเงินที่เมนูออเดอร์ออนไลน์แทน เพื่อไม่ให้สถานะสองฝั่งไม่ตรงกัน`,
      );
    }

    const productIds = [sale.productId, ...sale.bundleProductIds];

    // G7 — ใบซ่อมที่ยังเปิดอยู่บนเครื่องของใบขายนี้
    // RepairTicket ไม่มี saleId (ผูกกับ productId/contractId) จึงตรวจผ่าน productId
    const openTicket = await tx.repairTicket.findFirst({
      where: {
        productId: { in: productIds },
        deletedAt: null,
        status: { notIn: ['CLOSED', 'CANCELLED', 'REPLACED'] },
      },
      select: { ticketNumber: true, status: true },
    });
    if (openTicket) {
      throw new BadRequestException(
        `มีใบซ่อม ${openTicket.ticketNumber} (สถานะ ${openTicket.status}) ที่ยังไม่ปิดบนเครื่องของใบขายนี้ — ปิดหรือยกเลิกใบซ่อมก่อน`,
      );
    }

    // G5 — เครื่องทุกชิ้น (หลัก + ของแถม) ต้องยังอยู่ในสถานะที่ใบขายนี้ตั้งไว้เอง
    const expectedStatus = SALE_TYPE_PRODUCT_STATUS[sale.saleType as 'CASH' | 'EXTERNAL_FINANCE'];
    const products = await tx.product.findMany({
      where: { id: { in: productIds } },
      select: { id: true, name: true, status: true, deletedAt: true },
    });
    for (const p of products) {
      await assertProductNotHeld(tx, { ...p, expectedStatus }, 'RESTORE_TO_STOCK');
    }
    if (products.length !== productIds.length) {
      throw new BadRequestException('ไม่พบสินค้าบางรายการของใบขายนี้ — ข้อมูลผิดปกติ');
    }

    // G3 — ไฟแนนซ์ภายนอกโอนมาแล้ว
    if (sale.saleType === 'EXTERNAL_FINANCE') {
      const receivable = await tx.financeReceivable.findFirst({
        where: { saleId: sale.id, deletedAt: null },
        select: { id: true, status: true, receivedAmount: true, financeCompany: true },
      });
      if (
        receivable &&
        (receivable.status !== 'PENDING' ||
          (receivable.receivedAmount && receivable.receivedAmount.greaterThan(0)))
      ) {
        throw new BadRequestException(
          `ไฟแนนซ์ ${receivable.financeCompany} โอนเงินของใบขายนี้มาแล้ว (สถานะ ${receivable.status}) — ` +
            'ต้องบันทึกคืนเงิน/ปรับรายการรับจากไฟแนนซ์ให้เรียบร้อยก่อนจึงจะยกเลิกใบขายได้',
        );
      }
    }

    // G4 — ค่าคอมจ่ายแล้ว
    const paidCommission = await tx.salesCommission.findFirst({
      where: { saleId: sale.id, status: 'PAID', deletedAt: null },
      select: { id: true, period: true },
    });
    if (paidCommission) {
      throw new BadRequestException(
        `ค่าคอมของใบขายนี้จ่ายออกไปแล้ว (งวด ${paidCommission.period}) — ` +
          'ต้องเรียกคืน/ปรับงวดค่าคอมก่อนจึงจะยกเลิกใบขายได้',
      );
    }

    // G2 — งวดบัญชีของวันที่จะโพสต์กลับรายการ (เฉพาะใบที่มี JE)
    const saleJes = await tx.journalEntry.findMany({
      where: {
        metadata: { path: ['saleId'], equals: sale.id } as any,
        status: 'POSTED',
        deletedAt: null,
      },
      select: { id: true, companyId: true },
    });
    const now = new Date();
    for (const companyId of [...new Set(saleJes.map((j) => j.companyId))]) {
      await validatePeriodOpen(tx, now, companyId);
    }

    // ===== ผ่านด่านครบแล้ว เริ่มเขียน =====

    await tx.product.updateMany({
      where: { id: { in: productIds } },
      data: { status: ProductStatus.IN_STOCK },
    });

    let reversalEntryNumbers: string[] = [];
    if (saleJes.length) {
      const { reversalJeIds } = await this.reversalTemplate.reverse(
        {
          jeIds: [],
          sweepBy: { path: 'saleId', value: sale.id },
          flowLabel: 'shop-cash-sale-void',
          descriptionPrefix: '[ยกเลิกใบขาย]',
        },
        tx,
      );
      const created = await tx.journalEntry.findMany({
        where: { id: { in: reversalJeIds } },
        select: { entryNumber: true },
      });
      reversalEntryNumbers = created.map((c) => c.entryNumber);
    }

    if (sale.saleType === 'EXTERNAL_FINANCE') {
      await tx.financeReceivable.updateMany({
        where: { saleId: sale.id, deletedAt: null },
        data: { deletedAt: now },
      });
    } else {
      await tx.salesCommission.updateMany({
        where: { saleId: sale.id, status: { in: ['PENDING', 'APPROVED'] }, deletedAt: null },
        data: { status: 'CLAWED_BACK' },
      });
    }

    await tx.sale.update({
      where: { id: sale.id },
      data: { deletedAt: now, voidReason: reason, voidedById: userId },
    });

    await tx.auditLog.create({
      data: {
        action: 'SALE_VOIDED',
        entity: 'sale',
        entityId: sale.id,
        userId,
        newValue: {
          saleNumber: sale.saleNumber,
          saleType: sale.saleType,
          netAmount: sale.netAmount.toString(),
          reason,
          restoredProductIds: productIds,
          reversalEntryNumbers,
        },
      },
    });

    return { saleNumber: sale.saleNumber, restoredProductIds: productIds, reversalEntryNumbers };
  }
}
```

/** Dr − Cr ต่อบัญชี ของ JE ทุกใบที่เกี่ยวกับใบขายนี้ (ใบเดิม + ใบกลับรายการ) */
async function netByAccount(saleId: string, reversalJeIds: string[]) {
  const entries = await prisma.journalEntry.findMany({
    where: {
      OR: [
        { metadata: { path: ['saleId'], equals: saleId } as any },
        { id: { in: reversalJeIds } },
      ],
      status: 'POSTED',
      deletedAt: null,
    },
    include: { lines: true },
  });
  const net: Record<string, Decimal> = {};
  for (const e of entries) {
    for (const l of e.lines) {
      net[l.accountCode] = (net[l.accountCode] ?? new Decimal(0))
        .plus(l.debit ?? 0)
        .minus(l.credit ?? 0);
    }
  }
  return net;
}

// ในเคสหลัก:
const net = await netByAccount(sale.id, res.reversalEntryNumbers.length ? reversalJeIds : []);
expect(Object.keys(net).length).toBeGreaterThan(0); // มี JE จริง ไม่ใช่ผ่านเพราะไม่มีอะไรเลย
for (const [code, amount] of Object.entries(net)) {
  expect(amount.toFixed(2), `บัญชี ${code} ต้องสุทธิเป็นศูนย์`).toBe('0.00');
}
